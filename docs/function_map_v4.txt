   2a. handle_browser_yt_runtime(voice_local_line, browser)
   2b. handle_browser_runtime(voice_local_line, browser, registry)
   2c. fallback: handle_browser_yt_runtime("yt " + line, browser)
   2d. jesli brak dopasowania: voice command pomijana
3. handle_camera_command(line)
4. handle_module_runtime(line, api)
5. handle_project_runtime(line, api)
6. handle_tokens_runtime(line, api)
7. handle_about_runtime(line, cfg)
8. handle_http_runtime(line, api)

11. handle_logs_runtime(line, api, cfg)
12. handle_vcs_runtime(line, api)
13. handle_audio_runtime(line)
14. handle_core_dispatch(line, api)
15. handle_basic_cli(line)
16. handle_runtime_meta(line, api, cfg)
17. handle_runtime_controls(line, cfg, api)
18. handle_browser_yt_runtime(line, browser)
19. handle_browser_runtime(line, browser, registry)

21. handle_explicit_dispatch(line, api)
